# ArtDex Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build ArtDex — a "real-world masterpiece Pokédex" where users photograph artworks at museums, AI identifies them against a location-narrowed candidate set, and they collect them into a personal Dex with by-artist sets, rarity tiers, a world map, selfies, and visit notes.

> **⚠️ DB UPDATE (2026-06-12): shipped on Amazon DynamoDB, not Aurora.** The free-plan AWS account
> blocks RDS Data API on Aurora, so Phases 2/5 were implemented against DynamoDB instead:
> `src/lib/aws/dynamo.ts` (client), `scripts/seedDynamo.ts` (seed), `src/lib/db/queries.ts` (access),
> and the 6 routes under `src/app/api/`. No PostGIS — geo is done in-app. Tasks below that mention
> Aurora/`rdsData.ts`/`applySchema.ts`/Data API are superseded. **Auth** also changed: Clerk was
> specced but not adopted — a cookie-based anonymous id (`src/lib/auth.ts`) gives each browser its
> own collection. `TODO.md` has the live checklist.

**Architecture:** Next.js (App Router) on Vercel. Aurora PostgreSQL Serverless v2 + PostGIS, accessed over the RDS Data API (HTTP, serverless-friendly). Photo recognition via Claude vision on AWS Bedrock, made reliable by narrowing the candidate set to the works currently exhibited at the user's GPS-detected museum. Images in S3. Auth via Clerk. Map via react-leaflet + OpenStreetMap.

**Tech Stack:** TypeScript, Next.js 15 (App Router), Tailwind CSS, Vitest, Aurora PostgreSQL + PostGIS, `@aws-sdk/client-rds-data`, `@aws-sdk/client-bedrock-runtime`, `@aws-sdk/client-s3`, Clerk, react-leaflet, framer-motion, Zod.

## Conventions

- **TDD where it pays:** pure logic (rarity, distance/location-gate, candidate filtering, progress math, recognition-response parsing, Zod validation) gets a failing test first. AWS/I/O boundaries (Data API, Bedrock, S3) get run-and-verify steps (curl/script + DB check) instead of mocked unit tests.
- **Commit after every task.** Conventional commit messages (`feat:`, `fix:`, `chore:`, `docs:`).
- **Env vars** live in `.env.local` (never committed). A committed `.env.example` documents every key.
- **One responsibility per file.** Data access in `src/lib/db/`, AWS clients in `src/lib/aws/`, domain logic in `src/lib/domain/`, API routes in `src/app/api/`, UI in `src/components/` and route folders.

```ts
export function readEnv<K extends string>(
  source: Record<string, string | undefined>,
  keys: K[]
): Record<K, string> {
  const out = {} as Record<K, string>;
  for (const k of keys) {
    const v = source[k];
    if (!v) throw new Error(`Missing required env var: ${k}`);
    out[k] = v;
  }
  return out;
}
```

```ts
import {
  RDSDataClient,
  ExecuteStatementCommand,
  type Field,
  type ColumnMetadata,
} from "@aws-sdk/client-rds-data";
import { readEnv } from "@/lib/env";

function fieldValue(f: Field): unknown {
  if (f.isNull) return null;
  if (f.stringValue !== undefined) return f.stringValue;
  if (f.longValue !== undefined) return f.longValue;
  if (f.doubleValue !== undefined) return f.doubleValue;
  if (f.booleanValue !== undefined) return f.booleanValue;
  return null;
}

export function parseRecords(records: Field[][], cols: ColumnMetadata[]) {
  return records.map((row) => {
    const obj: Record<string, unknown> = {};
    row.forEach((f, i) => { obj[cols[i].name ?? String(i)] = fieldValue(f); });
    return obj;
  });
}

let _client: RDSDataClient | null = null;
function client() {
  const { AWS_REGION } = readEnv(process.env, ["AWS_REGION"]);
  return (_client ??= new RDSDataClient({ region: AWS_REGION }));
}

export type SqlParam = { name: string; value: Field };

export async function query<T = Record<string, unknown>>(
  sql: string,
  params: SqlParam[] = []
): Promise<T[]> {
  const env = readEnv(process.env, ["RDS_CLUSTER_ARN", "RDS_SECRET_ARN", "RDS_DATABASE"]);
  const res = await client().send(new ExecuteStatementCommand({
    resourceArn: env.RDS_CLUSTER_ARN,
    secretArn: env.RDS_SECRET_ARN,
    database: env.RDS_DATABASE,
    sql,
    parameters: params.map((p) => ({ name: p.name, value: p.value })),
    includeResultMetadata: true,
  }));
  return parseRecords(res.records ?? [], res.columnMetadata ?? []) as T[];
}

// helpers for building params
export const S = (name: string, v: string): SqlParam => ({ name, value: { stringValue: v } });
export const B = (name: string, v: boolean): SqlParam => ({ name, value: { booleanValue: v } });
export const D = (name: string, v: number): SqlParam => ({ name, value: { doubleValue: v } });
```

```ts
export const GATE_RADIUS_M = 150; // must be within 150m of the museum
export function haversineMeters(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const R = 6371000, toRad = (d: number) => (d * Math.PI) / 180;
  const dLat = toRad(lat2 - lat1), dLon = toRad(lon2 - lon1);
  const a = Math.sin(dLat / 2) ** 2 +
    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLon / 2) ** 2;
  return 2 * R * Math.asin(Math.sqrt(a));
}
export function isWithinGate(_unused: number, distanceM: number): boolean {
  return distanceM <= GATE_RADIUS_M;
}
```

```ts
export function buildRecognitionPrompt(c: { id: string; title: string; artist: string }[]): string {
  const list = c.map((x) => `- ${x.id}: "${x.title}" by ${x.artist}`).join("\n");

    `Reply with ONLY the matching id from the list, or "none" if none match.`;

export function parseRecognition(text: string, validIds: string[]): string | null {
  const t = text.trim().toLowerCase();
  return validIds.find((id) => t.includes(id.toLowerCase())) ?? null;
}
```

```ts
export async function recognizeArtwork(
  imageBase64: string, mediaType: string, prompt: string

  const client = new BedrockRuntimeClient({ region: AWS_REGION });
  const body = {
    anthropic_version: "bedrock-2023-05-31",
    max_tokens: 50,
    messages: [{ role: "user", content: [
      { type: "image", source: { type: "base64", media_type: mediaType, data: imageBase64 } },
      { type: "text", text: prompt },
    ]}],
  };
  const res = await client.send(new InvokeModelCommand({
    modelId: BEDROCK_MODEL_ID, contentType: "application/json", body: JSON.stringify(body),
  }));
  const json = JSON.parse(new TextDecoder().decode(res.body));
  return json.content?.[0]?.text ?? "";
}
```

```ts
            ST_Distance(geom, ST_SetSRID(ST_MakePoint(:lon,:lat),4326)::geography) AS dist_m

export async function currentCandidates(museumId: string) {
```

```ts
import { currentCandidates } from "@/lib/db/queries";
import { buildRecognitionPrompt, parseRecognition } from "@/lib/domain/recognition";
import { recognizeArtwork } from "@/lib/aws/bedrock";

export async function POST(req: NextRequest) {
  const { museumId, imageBase64, mediaType } = await req.json();
  if (!museumId || !imageBase64) return NextResponse.json({ error: "bad request" }, { status: 400 });
  const candidates = await currentCandidates(museumId);
  const prompt = buildRecognitionPrompt(candidates.map((c) => ({ id: c.id, title: c.title, artist: c.artist })));
  const text = await recognizeArtwork(imageBase64, mediaType ?? "image/jpeg", prompt);
  const id = parseRecognition(text, candidates.map((c) => c.id));
  const artwork = candidates.find((c) => c.id === id) ?? null;
  return NextResponse.json({ artwork });
}
```

### Task 5.5: `/api/collect` (location-gated)

**Files:** Create `src/app/api/collect/route.ts`.

- [ ] **Step 1 (impl):** POST `{ artworkId, museumId, lat, lon, selfieUrl?, photoUrl?, note? }`. Steps: `requireUserId()`; `getArtwork`; if `isOnSiteRequired(rarity)` → load museum geom, compute `haversineMeters`, reject if `!isWithinGate`; insert into `collections` (ON CONFLICT do nothing); build `exhibition_label` from museum; return `{ collected: true, rarity }`.

```ts
import { NextRequest, NextResponse } from "next/server";
import { requireUserId } from "@/lib/auth";
import { getArtwork } from "@/lib/db/queries";
import { query, S, D } from "@/lib/aws/rdsData";
import { isOnSiteRequired, type Rarity } from "@/lib/domain/rarity";
import { haversineMeters, isWithinGate } from "@/lib/domain/locationGate";
import { randomUUID } from "node:crypto";

export async function POST(req: NextRequest) {
  const userId = await requireUserId();
  const b = await req.json();
  const art = await getArtwork(b.artworkId);
  if (!art) return NextResponse.json({ error: "unknown artwork" }, { status: 404 });

  if (isOnSiteRequired(art.rarity as Rarity)) {
    const m = await query<{ name: string; city: string; lat: number; lon: number }>(
      `SELECT name, city, ST_Y(geom::geometry) AS lat, ST_X(geom::geometry) AS lon
       FROM museums WHERE id = :id`, [S("id", b.museumId)]
    );
    if (!m[0]) return NextResponse.json({ error: "museum required for legendary" }, { status: 400 });
    const dist = haversineMeters(b.lat, b.lon, m[0].lat, m[0].lon);
    if (!isWithinGate(0, dist))
      return NextResponse.json({ error: "must be on-site to collect this legendary", distance: dist }, { status: 403 });
  }

  const label = b.museumId
    ? (await query<{ name: string; city: string }>(`SELECT name, city FROM museums WHERE id=:id`, [S("id", b.museumId)]))[0]
    : null;
  await query(
    `INSERT INTO collections(id,user_id,artwork_id,museum_id,exhibition_label,selfie_url,photo_url,note)
     VALUES(:id,:uid,:aid,:mid,:label,:selfie,:photo,:note)
     ON CONFLICT (user_id, artwork_id) DO NOTHING`,
    [S("id", randomUUID()), S("uid", userId), S("aid", b.artworkId),
     S("mid", b.museumId ?? ""), S("label", label ? `${label.name}, ${label.city}` : ""),
     S("selfie", b.selfieUrl ?? ""), S("photo", b.photoUrl ?? ""), S("note", b.note ?? "")]
  );
  return NextResponse.json({ collected: true, rarity: art.rarity });
}
```
